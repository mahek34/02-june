2.Create a constant variable to store the GST rate (for example, 18%) and use it to calculate the final price of a Zomato order with a given base price.The GST rate must not be changeable after its initial assignment.
ANSWER:#include <stdio.h>
int main()
{
    const float GST = 18.0;
    float price = 500;
    float finalPrice;

    finalPrice = price + (price * GST / 100);

    printf("Final Price = %.2f", finalPrice);

    return 0;
}
